Función 1 · // (división entera)

Función 2 · % (residuo)

def f2(a, b):

Función 3 · / (siempre da decimal)

def f3(a, b):
    return a / b

Función 4 · ** (potencia)

def f4(a, b):
    return a ** b

Función 5 · primero * y luego +

def f5(x):

Función 6 · // y % juntos

def f6(n):
    return n // 10 + n % 10

Función 7 · de izquierda a derecha

def f7(a, b):
    return a // b * b

Función 8 · % antes que -

def f8(x):
    return 10 - x % 3

Función 9 · % 2 para saber si es par

def f9(n):
    if n % 2 == 0:
        return "par"
    else:
        return "impar"

Función 10 · if y una resta

def f10(precio):
    if precio >= 100:
        return precio - 20
    return precio
