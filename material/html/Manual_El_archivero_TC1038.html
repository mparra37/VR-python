El archivero

El Operador ve dos listas paralelas, nombres y prom, como cajas con su posición (0, 1, 2…). Ustedes tienen el código pero no los datos: pídanselos por posición ("¿qué promedio hay en la posición 2?"). El Operador les lee la pregunta; ustedes siguen el código y le dicen la respuesta. No se usan sum(), max() ni sort().

buscar

def buscar(nombres, x):
    for i in range(len(nombres)):
        if nombres[i] == x:
            return i
    return -1

promedio_grupo

def promedio_grupo(prom):
    total = 0
    for p in prom:
        total += p
    return round(total / len(prom), 2)

mejor

def mejor(nombres, prom):
    m = 0
    for i in range(len(prom)):
        if prom[i] > prom[m]:
            m = i
    return nombres[m]

aprobados

def aprobados(nombres, prom):
    lista = []
    for i in range(len(prom)):
        if prom[i] >= 70:
            lista.append(nombres[i])
    return len(lista)

reprobados

def reprobados(nombres, prom):
    lista = []
    for i in range(len(prom)):
        if prom[i] < 70:
            lista.append(nombres[i])
    return len(lista)

eliminar

def eliminar(nombres, prom, x):
    pos = buscar(nombres, x)
    nombres.pop(pos)
    prom.pop(pos)

lugar

def lugar(prom, i):
    l = 1
    for p in prom:
        if p > prom[i]:
            l += 1
    return l

agregar

def agregar(nombres, prom, x, p):
    nombres.append(x)
    prom.append(p)
    return len(nombres) - 1
